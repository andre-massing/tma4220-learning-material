% The first article (the kick-off lecture) is typeset as an unnumbered
% chapter, so that the substantive chapters start at 1. MyST emits a plain
% \chapter{...} for every article, so \chapter is wrapped in a one-shot
% toggle: the first call becomes \chapter*, is added to the table of
% contents by hand, and has section numbering switched off; every later
% call restores normal numbering and behaves as usual.
\newif\ifmystfirstchapter
\mystfirstchaptertrue
\let\mystorigchapter\chapter
% Signature "s m": the starred form (\chapter*{...}, used by the bibliography)
% must pass through untouched. A plain \renewcommand{\chapter}[1] would swallow
% the star and typeset a numbered chapter titled "*".
% (No optional argument is declared: MyST never emits a short-title form, and
% square-bracket-hash sequences in this file collide with the template engine.)
\RenewDocumentCommand{\chapter}{s m}{%
  \IfBooleanTF{#1}{%
    \mystorigchapter*{#2}%
  }{%
    \ifmystfirstchapter
      \mystfirstchapterfalse
      \setcounter{secnumdepth}{-1}%
      \mystorigchapter*{#2}%
      \addcontentsline{toc}{chapter}{#2}%
      \markboth{#2}{}%
    \else
      \setcounter{secnumdepth}{2}%
      \mystorigchapter{#2}%
    \fi
  }%
}
